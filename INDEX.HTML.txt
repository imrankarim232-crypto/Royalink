        // Gallery Filter
        function filterGallery(category) {
            const items = document.querySelectorAll('.gallery-item');
            const buttons = document.querySelectorAll('.gallery-btn');

            buttons.forEach(btn => {
                btn.classList.remove('bg-gold', 'text-black');
                btn.classList.add('bg-cardBg', 'text-gray-300');
            });
            event.target.classList.remove('bg-cardBg', 'text-gray-300');
            event.target.classList.add('bg-gold', 'text-black');

            items.forEach(item => {
                if (category === 'all' || item.dataset.category === category) {
                    item.style.display = 'block';
                } else {
                    item.style.display = 'none';
                }
            });
        }

        function preselectStyle(styleName) {
            const select = document.getElementById('booking-style');
            for(let i=0; i<select.options.length; i++) {
                if(select.options[i].text.includes(styleName) || select.options[i].value.includes(styleName)) {
                    select.selectedIndex = i;
                    break;
                }
            }
            window.location.href = '#booking';
        }

        // Price Estimator
        function calculateEstimate() {
            const styleMultiplier = parseFloat(document.getElementById('est-style').value);
            const size = parseInt(document.getElementById('est-size').value);
            const currency = document.getElementById('est-currency').value;

            let baseMin = 150 + (size * 35 * styleMultiplier);
            let baseMax = 300 + (size * 60 * styleMultiplier);
            let hoursMin = Math.max(1, Math.round(size / 3));
            let hoursMax = hoursMin + 2;
            let deposit = 50;

            let rate = 1;
            let symbol = '$';
            if (currency === 'KES') { rate = 130; symbol = 'KES '; }
            else if (currency === 'EUR') { rate = 0.92; symbol = '€'; }
            else if (currency === 'GBP') { rate = 0.78; symbol = '£'; }
            else if (currency === 'AUD') { rate = 1.52; symbol = 'A$'; }
            else if (currency === 'JPY') { rate = 155; symbol = '¥'; }

            const finalMin = Math.round(baseMin * rate);
            const finalMax = Math.round(baseMax * rate);
            const finalDeposit = Math.round(deposit * rate);

            document.getElementById('size-val').innerText = size + ' inches';
            document.getElementById('result-price').innerText = symbol + finalMin.toLocaleString() + ' - ' + symbol + finalMax.toLocaleString();
            document.getElementById('result-time').innerText = hoursMin + ' - ' + hoursMax + ' Hours';
            document.getElementById('result-deposit').innerText = symbol + finalDeposit.toLocaleString();
        }

        function transferEstimateToBooking() {
            window.location.href = '#booking';
        }

            for(let i=1; i<=4; i++) {

                indicator.className = "text-gray-500 flex items-center gap-1.5";
                indicator.children[0].className = "w-6 h-6 rounded-full bg-gray-800 text-gray-400 flex items-center justify-center font-bold";

                indicator.children[0].className = "w-6 h-6 rounded-full bg-gold text-black flex items-center justify-center font-bold";
            }
        }
